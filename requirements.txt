# LISA complete prerequisite manifest
#
# This file is intentionally pip-compatible: only the uncommented requirement
# lines are installed by pip. System software and cloud prerequisites are
# documented below and installed or checked by Install-LISA-Prerequisites.ps1.
#
# Supported host: Windows 11
# Python: 3.11 or newer; CPython 3.13.15 verified
# Install Python packages with:
#   python -m pip install --upgrade -r requirements.txt

# JSON Schema validation is shared by all lifecycle stages. The non-GPL format
# extra guarantees that URI and RFC 3339 date-time checks used by the packaged
# schemas are active instead of being silently skipped.
jsonschema[format-nongpl]>=4.23,<5

# Requirement Analyzer document and image extraction.
pypdf>=6.0,<7
python-docx>=1.2,<2
python-pptx>=1.0,<2
Pillow>=11,<13

# The analyzer records this version in its extraction fingerprint and the test
# suite uses it to construct XLSX fixtures. Runtime XLSX scanning is streamed.
openpyxl>=3.1,<4

# Windows Python does not ship the IANA database used by ZoneInfo. LISA accepts
# IANA names such as Asia/Kolkata in lisa-config.json, so install it everywhere
# to make fresh-machine behavior deterministic.
tzdata>=2024.1

# Solution Designer uses NetworkX for deterministic obstacle-aware orthogonal
# routing. Pin it because the installed version participates in cache identity.
networkx==3.6.1

#

# PowerShell
#   - PowerShell 7+: WinGet package Microsoft.PowerShell
#   - Windows PowerShell 5.1: built into supported Windows installations and

#     solution packaging. Required groups include auth, env, org, copilot, and

#   - Do not substitute the legacy WinGet Microsoft.PowerAppsCLI 1.0 package.

# .NET 10 SDK
#   - WinGet package: Microsoft.DotNet.SDK.10
#   - Required when bootstrapping modern pac as a .NET tool.
